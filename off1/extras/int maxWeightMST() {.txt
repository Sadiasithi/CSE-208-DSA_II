int maxWeightMST() {
        Arrays.sort(edges); // Sort edges in descending order of weight

        int[] parent = new int[V];
        Arrays.fill(parent, -1);

        int maxWeight = 0;
        for (int i = 0; i < E; i++) {
            Edge currentEdge = edges[i];
            int x = find(parent, currentEdge.src);
            int y = find(parent, currentEdge.dest);

            if (x != y) {
                maxWeight += currentEdge.weight;
                union(parent, x, y);
            }
        }

        return maxWeight;
    }
